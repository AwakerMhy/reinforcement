# 有界标签自然配对复现程序（JavaScript）

可在 JavaScript V8/Node 直接执行；定义共享准备与续训标签 tape、自然有序入组、最少计数/固定 A 对照。结果对应共享环境独立复核。

## 可运行源代码

```javascript
function rng(seed){let x=seed>>>0;return()=>{x^=x<<13;x^=x>>>17;x^=x<<5;return(x>>>0)/4294967296}}
function bit(seed,j,k){let x=(seed+Math.imul(j+1,0x9e3779b9)+Math.imul(k+1,0x85ebca6b))>>>0;x^=x>>>16;x=Math.imul(x,0x7feb352d);x^=x>>>15;x=Math.imul(x,0x846ca68b);return(x^x>>>16)&1}
function risk(v){let[a,b,sa,sb]=v,x=(sa+1)/(a+2)-.5,y=(sb+1)/(b+2)-.5;return x*x+.001*y*y}
function prep(r,p,seed){let v=[0,0,0,0];for(let k=0;k<20;k++){let j=r()<p?0:1;v[j+2]+=bit(seed,j,v[j]);v[j]++}return v}
function hit(v,seed,mode){v=v.slice();if(risk(v)<=.01)return 0;let c=[0,0];for(let t=1;t<=1000;t++){let j=mode===0?(v[0]<=v[1]?0:1):0;v[j+2]+=bit(seed,j,c[j]++);v[j]++;if(risk(v)<=.01)return t}return 1001}
let N=100000,stats=[0,1].map(()=>({n:0,strong:0,weak:0,gap:0,gapSq:0,weakFirst:0,censor:0})),initialS=0,initialW=0;
for(let i=0;i<N;i++){
  let off=Math.imul(i+1,7919),labelSeed=(0xe3300000+off)>>>0;
  let S=prep(rng((0xe1100000+off)>>>0),.8,labelSeed),W=prep(rng((0xe2200000+off)>>>0),.2,labelSeed);
  let s=risk(S),w=risk(W);if(!(s>.01&&w>s))continue;initialS+=s;initialW+=w;
  let future=(0xe4400000+off)>>>0;
  for(let mode=0;mode<2;mode++){
    let ts=hit(S,future,mode),tw=hit(W,future,mode),d=ts-tw,q=stats[mode];
    q.n++;q.strong+=ts;q.weak+=tw;q.gap+=d;q.gapSq+=d*d;q.weakFirst+=tw<ts;q.censor+=ts>1000||tw>1000;
  }
}
console.log({selected:stats[0].n,initial:[initialS/stats[0].n,initialW/stats[0].n],results:stats.map(q=>({strong:q.strong/q.n,weak:q.weak/q.n,gap:q.gap/q.n,se:Math.sqrt((q.gapSq/q.n-(q.gap/q.n)**2)/q.n),weakFirst:q.weakFirst/q.n,censor:q.censor}))});
```

运行输出应为：selected=16654；最少计数强/弱35.114507/16.742945、差18.371562；固定A强/弱12.583223/11.571815、差1.011409。
