# UCI Wine 主动学习五划分复算脚本

与[五划分负结果](https://chatgpt.com/space/page_309fe32f08308191b7a42ba745235a99)对应。代码使用 Node.js 全局 fetch 从 UCI 官方 CSV 读取数据；不需第三方包。

```js
const URL="https://archive.ics.uci.edu/ml/machine-learning-databases/wine-quality/winequality-red.csv";
function rng(seed){let x=seed>>>0;return()=>{x+=0x6D2B79F5;let t=x;t=Math.imul(t^t>>>15,t|1);t^=t+Math.imul(t^t>>>7,t|61);return ((t^t>>>14)>>>0)/4294967296}}
function sig(x){return x>=0?1/(1+Math.exp(-x)):Math.exp(x)/(1+Math.exp(x))}
function sp(x){return Math.max(0,x)+Math.log1p(Math.exp(-Math.abs(x)))}
function dot(w,x){let s=0;for(let j=0;j<12;j++)s+=w[j]*x[j];return s}
function risk(w,d){let s=0;for(let z of d)s+=sp((1-2*z.y)*dot(w,z.x));return s/d.length}
function step(w,z){let g=.05*(sig(dot(w,z.x))-z.y);for(let j=0;j<12;j++)w[j]-=g*z.x[j]}
async function main(){
  let csv=await (await fetch(URL)).text();
  let raw=csv.trim().split(/\r?\n/).slice(1).map((line,i)=>{let a=line.split(";").map(Number);return {i:i+1,x:a.slice(0,11),y:a[11]>=6?1:0}});
  if(raw.length!==1599||raw.some(z=>z.x.length!==11||z.x.some(v=>!Number.isFinite(v))))throw Error("Unexpected Wine CSV");
  let all=[];
  for(let rep=0;rep<5;rep++){
    let r=rng(778+rep),by=[raw.filter(z=>z.y===0).map(z=>({...z,x:[...z.x]})),raw.filter(z=>z.y===1).map(z=>({...z,x:[...z.x]}))];
    for(let a of by)for(let i=a.length-1;i>0;i--){let j=Math.floor(r()*(i+1));[a[i],a[j]]=[a[j],a[i]]}
    let train=by.flatMap(a=>a.slice(0,Math.floor(a.length*.6)));
    let val=by.flatMap(a=>a.slice(Math.floor(a.length*.6),Math.floor(a.length*.8)));
    let test=by.flatMap(a=>a.slice(Math.floor(a.length*.8)));
    let mu=Array.from({length:11},(_,j)=>train.reduce((s,z)=>s+z.x[j],0)/train.length);
    let sd=mu.map((m,j)=>Math.sqrt(train.reduce((s,z)=>s+(z.x[j]-m)**2,0)/train.length));
    for(let z of train.concat(val,test))z.x=[1,...z.x.map((v,j)=>Math.max(-3,Math.min(3,(v-mu[j])/sd[j])))];
    let pp=[train.filter(z=>z.y===0),train.filter(z=>z.y===1)],P={S:[],W:[]};
    for(let i=0;i<20000;i++){
      let q=rng(9000000+rep*100000+i),w=Array.from({length:12},()=>.1*(q()+q()+q()+q()+q()+q()-3)),k=i%2?8:3;
      for(let t=0;t<10;t++){let pool=pp[t<k?0:1];step(w,pool[Math.floor(q()*pool.length)])}
      let v=risk(w,val);
      if(k===3&&v>=.63&&v<.64)P.S.push({w,Lv:v,Lt:risk(w,test)});
      if(k===8&&v>=.67&&v<.68)P.W.push({w,Lv:v,Lt:risk(w,test)});
    }
    let rr=rng(10000000+rep),T=[[],[],[],[]],strict=0;
    for(let pair=0;pair<100;pair++){
      let a=P.S[Math.floor(rr()*P.S.length)],b=P.W[Math.floor(rr()*P.W.length)];
      if(b.Lt>a.Lt)strict++;
      let ws=[a.w.slice(),b.w.slice(),a.w.slice(),b.w.slice()],h=[-1,-1,-1,-1],q=rng(11000000+rep*100000+pair);
      for(let t=1;t<=500;t++){
        if(h.every(x=>x>=0))break;
        let slate=Array.from({length:10},()=>train[Math.floor(q()*train.length)]),u=q(),v=q(),jr=Math.floor(q()*10);
        for(let g=0;g<4;g++){
          if(h[g]>=0)continue;
          let j=jr;
          if(g<2){if(u<.4)j=Math.floor(v*10);else{let best=Infinity;for(let m=0;m<10;m++){let d=Math.abs(dot(ws[g],slate[m].x));if(d<best){best=d;j=m}}}}
          step(ws[g],slate[j]);if(risk(ws[g],test)<=.60)h[g]=t;
        }
      }
      for(let g=0;g<4;g++)T[g].push(h[g]);
    }
    all.push({rep,pool:[P.S.length,P.W.length],strictPairs:strict,T:T.map(x=>x.reduce((s,z)=>s+(z<0?500:z),0)/x.length),censor:T.map(x=>x.filter(z=>z<0).length)});
  }
  console.log(all);
}
main().catch(console.error);
```
