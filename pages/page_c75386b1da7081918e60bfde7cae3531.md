# 三步 MDP 独立验证：最小 Node.js 复算脚本

与[结果页](https://chatgpt.com/space/page_e528e033a7f88191ae960d50403f069e)对应。保存为 `rl3_verify.js`，运行 `node rl3_verify.js`。不需外部包。目标输出约为 S n6090、T140.35±1.12、截尾1；W n5944、T32.87±.20、截尾0。

```js
const N=300000, PRE=6, MAX=300, pB=.3, temp=.15, alpha=.5, target=.85;
function rng(v){let s=v>>>0;return()=>{s+=0x6D2B79F5;let t=s;t=Math.imul(t^t>>>15,t|1);t^=t+Math.imul(t^t>>>7,t|61);return ((t^t>>>14)>>>0)/4294967296}}
function pi(q){return 1/(1+Math.exp((q[1]-q[0])/temp))}
function init(){let q={};for(let c of ["A","B"]){q[c+"0"]=[0,0];for(let a=0;a<2;a++){q[c+"1"+a]=[0,0];for(let b=0;b<2;b++)q[c+"2"+a+b]=[0,0]}}return q}
function score(q){return (1-pB)*pi(q.A0)+pB*pi(q.B0)*pi(q.B10)*pi(q.B200)}
function episode(q,r){
  let B=r()<pB,c=B?"B":"A",a0=r()<pi(q[c+"0"])?0:1,n1=c+"1"+a0;
  q[c+"0"][a0]+=alpha*(Math.max(...q[n1])-q[c+"0"][a0]);
  let a1=r()<pi(q[n1])?0:1,n2=c+"2"+a0+a1;
  q[n1][a1]+=alpha*(Math.max(...q[n2])-q[n1][a1]);
  let a2=r()<pi(q[n2])?0:1;
  let reward=B?(a0===0&&a1===0&&a2===0?1:0):(a0===0?1:0);
  q[n2][a2]+=alpha*(reward-q[n2][a2]);
  return B;
}
const G={S:{n:0,j:0,t:0,t2:0,censored:0},W:{n:0,j:0,t:0,t2:0,censored:0}};
for(let i=0;i<N;i++){
  let r=rng(15000000+i),q=init(),k=0;
  for(let t=0;t<PRE;t++)k+=episode(q,r);
  let j=score(q);
  let group=k===0&&q.A0[0]>0&&j>=.55&&j<.70?"S":
    (k===2||k===3)&&q.A0[0]===0&&q.B10[0]>0&&q.B0[0]===0&&j>=.46&&j<.50?"W":null;
  if(!group)continue;
  let t=0;while(t<MAX&&score(q)<target){episode(q,r);t++}
  let x=3*t,g=G[group];g.n++;g.j+=j;g.t+=x;g.t2+=x*x;g.censored+=+(t===MAX&&score(q)<target);
}
for(let [group,g] of Object.entries(G)){
  let mean=g.t/g.n,se=Math.sqrt((g.t2-g.t*g.t/g.n)/(g.n-1)/g.n);
  console.log(group,{n:g.n,meanInitialJ:g.j/g.n,meanExtraInteractions:mean,SE:se,censored:g.censored});
}
```

注意：脚本复算独立初态验证，配对共用未来随机流与随机转移扩展见结果页的种子和规则；准备阶段6回合不计入额外首达时间，但两组预算相同。
