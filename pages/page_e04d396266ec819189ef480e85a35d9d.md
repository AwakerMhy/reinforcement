# 两步 MDP 同环境首达效应复现程序

Node.js 或 JavaScript V8 可运行；包含确定转移、随机转移和第二状态Q值消融的全部预定随机种子。

## 完整复现代码

```javascript
function main(){
function U(seed,t,c){let x=(seed+Math.imul(t+1,0x9e3779b9)+Math.imul(c+1,0x85ebca6b))>>>0;x^=x>>>16;x=Math.imul(x,0x7feb352d);x^=x>>>15;x=Math.imul(x,0x846ca68b);return((x^x>>>16)>>>0)/4294967296}
function value(q,pB){return q[0]>=q[1]?.6:(q[3]>=q[2]?pB*.2:pB*.95)}
function step(q,a,d,t,seed,pB){if(!a){q[0]+=.2*((U(seed,t,4)<.6?1:0)-q[0]);return}let reach=pB===1||U(seed,t,5)<pB;q[1]+=.2*((reach?Math.max(q[2],q[3]):0)-q[1]);if(!reach)return;let j=d?3:2;q[j]+=.2*((U(seed,t,j+4)<(d?.2:.95)?1:0)-q[j])}
function prep(choice,env,p,pB){let q=[0,0,0,0];for(let t=0;t<30;t++)step(q,U(choice,t,0)<p?0:1,U(choice,t,1)<.95,t,env,pB);return q}
function hit(q,env,pB,epsilon){q=q.slice();let goal=pB===1?.9:.7;for(let t=0;t<5000;t++){let a=U(env,t,0)<epsilon?(U(env,t,1)<.5?0:1):(q[0]>=q[1]?0:1),d=U(env,t,2)<epsilon?U(env,t,3)<.5:q[3]>=q[2];step(q,a,d,t,env,pB);if(value(q,pB)>=goal)return t+1}return 5001}
function experiment(base,pB,epsilon,ablation){let z={n:0,strong:0,weak:0,gap:0,gapSq:0,weakFirst:0,censor:0};for(let i=0;i<20000;i++){let off=Math.imul(i+1,7919),r=(base[2]+off)>>>0,S=prep((base[0]+off)>>>0,r,.9,pB),W=prep((base[1]+off)>>>0,r,.1,pB);if(!(value(S,pB)===.6&&value(W,pB)===pB*.2))continue;if(ablation>=1)S[2]=W[2]=0;if(ablation>=2)S[3]=W[3]=0;let f=(base[3]+off)>>>0,ts=hit(S,f,pB,epsilon),tw=hit(W,f,pB,epsilon),d=ts-tw;z.n++;z.strong+=ts;z.weak+=tw;z.gap+=d;z.gapSq+=d*d;z.weakFirst+=tw<ts;z.censor+=ts>5000||tw>5000}return{n:z.n,mean:[z.strong/z.n,z.weak/z.n],gap:z.gap/z.n,se:Math.sqrt((z.gapSq/z.n-(z.gap/z.n)**2)/z.n),weakFirst:z.weakFirst/z.n,censor:z.censor}}
return{det:experiment([0xc1100000,0xc2200000,0xc3300000,0xc4400000],1,.1,0),stoch:experiment([0xe1100000,0xe2200000,0xe3300000,0xe4400000],.8,.1,0),ablation:[0,1,2].map(k=>experiment([0xf1100000,0xf2200000,0xf3300000,0xf4400000],1,.1,k))};
}
console.log(main());
```
