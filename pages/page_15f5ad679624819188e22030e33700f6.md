# Beta–Bernoulli 总体准备分布精确枚举代码

JavaScript V8/Node 可运行；枚举1771个准备态和全部条件配对，精确计算准备分布上的平均风险曲线。

## 完整代码

```javascript
function main(){
function C(n,k){let x=1;for(let i=1;i<=k;i++)x=x*(n-i+1)/i;return x}
function states(p){let out=[];for(let n=0;n<=20;n++)for(let sa=0;sa<=n;sa++)for(let sb=0;sb<=20-n;sb++){let b=20-n,L=((sa+1)/(n+2)-.5)**2+.001*((sb+1)/(b+2)-.5)**2,prob=C(20,n)*p**n*(1-p)**b*C(n,sa)*C(b,sb)/2**20;out.push({n,sa,sb,L,prob})}return out}
function futureRisk(z,t){let a=z.n,b=20-a,ca=0,cb=0;for(let k=0;k<t;k++){if(a+ca<=b+cb)ca++;else cb++}return ((z.sa-a/2)**2+ca/4)/(a+ca+2)**2+.001*((z.sb-b/2)**2+cb/4)/(b+cb+2)**2}
function report(mode){let S=states(.8),W=states(.2),ps=S.map(()=>0),pw=W.map(()=>0),mass=0;for(let i=0;i<S.length;i++)for(let j=0;j<W.length;j++){let s=S[i].L,w=W[j].L,yes=mode==="bands"?(s>=.02&&s<.05&&w>=.08&&w<.15):(s>.01&&w>s);if(yes){let q=S[i].prob*W[j].prob;ps[i]+=q;pw[j]+=q;mass+=q}}let curve=[];for(let t=0;t<=100;t++){let s=0,w=0;for(let i=0;i<S.length;i++)if(ps[i])s+=ps[i]*futureRisk(S[i],t);for(let i=0;i<W.length;i++)if(pw[i])w+=pw[i]*futureRisk(W[i],t);curve.push([s/mass,w/mass])}return{mode,mass,cross:curve.findIndex(x=>x[0]>x[1]),target:[curve.findIndex(x=>x[0]<=.01),curve.findIndex(x=>x[1]<=.01)],points:[0,8,9,20].map(t=>[t,...curve[t]])}}
return[report("natural"),report("bands")];
}
console.log(main());
```
