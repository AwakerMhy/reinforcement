# UCI Wine 主动学习复现实验代码

Node.js 18+ 脚本；从 UCI 原始 CSV 下载红葡萄酒数据，复算第二批独立确认的两个查询规则。核心模拟已在执行环境用原始1599行数据运行并核对输出。

## 完整代码

保存为 `wine_mpemba.mjs` 后用 `node wine_mpemba.mjs` 运行。

```javascript
const url="https://archive.ics.uci.edu/ml/machine-learning-databases/wine-quality/winequality-red.csv";
const csv=await (await fetch(url)).text();
const rows=csv.trim().split(/\r?\n/).slice(1).map(line=>line.split(";").map(Number));
if(rows.length!==1599||rows.some(r=>r.length!==12))throw Error("Unexpected UCI data shape");
function experiment(rows){
const N=20000,eps=.03,cap=500;
function rng(s){let x=s>>>0;return()=>{x^=x<<13;x^=x>>>17;x^=x<<5;return(x>>>0)/4294967296}}
function shuffle(n,r){let a=Array.from({length:n},(_,i)=>i);for(let i=n-1;i>0;i--){let j=Math.floor(r()*(i+1));[a[i],a[j]]=[a[j],a[i]]}return a}
function error(v,m,w){return w[0]*(v[2]/(v[0]+2)-m[0])**2+w[1]*(v[3]/(v[1]+2)-m[1])**2}
function prep(p,r,t){let v=[0,0,11,11];for(let k=0;k<20;k++){let j=r()<p?0:1;v[2+j]+=t[j][v[j]];v[j]++}return v}
function hit(v,t,m,w,mode){v=v.slice();let c=[20,20];for(let k=1;k<=cap;k++){let j=mode===0?(v[0]<=v[1]?0:1):(k-1)%2;v[2+j]+=t[j][c[j]++];v[j]++;if(error(v,m,w)<=eps)return k}return cap+1}
let z=[0,1].map(()=>({n:0,s:0,w:0,d:0,q:0,wf:0,censor:0}));
for(let i=0;i<N;i++){
 let off=Math.imul(i+1,7919),idx=shuffle(rows.length,rng((0x61100000+off)>>>0)),tr=[[],[]],te=[[],[]];
 for(let k=0;k<idx.length;k++){let r=rows[idx[k]],j=r[10]>=10?0:1;(k<1279?tr:te)[j].push(r[11])}
 let m=te.map(x=>x.reduce((a,b)=>a+b,0)/x.length),w=te.map(x=>x.length/320);
 let a=prep(.8,rng((0x62200000+off)>>>0),tr),b=prep(.2,rng((0x63300000+off)>>>0),tr);
 if(!(error(a,m,w)>eps&&error(b,m,w)>error(a,m,w)))continue;
 for(let mode=0;mode<2;mode++){let ta=hit(a,tr,m,w,mode),tb=hit(b,tr,m,w,mode),d=ta-tb,q=z[mode];q.n++;q.s+=ta;q.w+=tb;q.d+=d;q.q+=d*d;q.wf+=tb<ta;q.censor+=ta>cap||tb>cap}
}
return z.map((q,k)=>({mode:k?"alternating":"minCount",n:q.n,Tcap:[q.s/q.n,q.w/q.n],gap:q.d/q.n,se:Math.sqrt((q.q/q.n-(q.d/q.n)**2)/q.n),weakFirst:q.wf/q.n,censor:q.censor}));
}
console.log(experiment(rows));
```
