## 被动有限宽 MLP：300,000 种子独立验证

```js
const N=300000,H=4,PRE=10,MAX=400,s=.25,eta=.3,target=.08;
function rng(v){let x=v>>>0;return()=>{x+=0x6D2B79F5;let t=x;t=Math.imul(t^(t>>>15),t|1);t^=t+Math.imul(t^(t>>>7),t|61);return((t^(t>>>14))>>>0)/4294967296}}
function gauss(r){return Math.sqrt(-2*Math.log(Math.max(r(),1e-12)))*Math.cos(2*Math.PI*r())}
function init(r){return Array.from({length:H},()=>[.5*gauss(r),.5*gauss(r),.5*gauss(r)])}
function f(q,b){let z=0,t=[];for(let h=0;h<H;h++){let y=Math.tanh(b?s*q[h][1]:q[h][0]);t.push(y);z+=q[h][2]*y}return{z:z/2,t}}
function comp(q){let a=f(q,0).z-1,b=f(q,1).z-1;return [.25*a*a,.25*b*b]}
function one(q,b){let o=f(q,b),g=eta*(o.z-1)/2;for(let h=0;h<H;h++){let v=q[h][2],t=o.t[h];q[h][b?1:0]-=g*v*(1-t*t)*(b?s:1);q[h][2]-=g*t}}
function both(q){let a=f(q,0),b=f(q,1),ga=eta*.25*(a.z-1),gb=eta*.25*(b.z-1);for(let h=0;h<H;h++){let v=q[h][2],ta=a.t[h],tb=b.t[h];q[h][0]-=ga*v*(1-ta*ta);q[h][1]-=gb*v*(1-tb*tb)*s;q[h][2]-=ga*ta+gb*tb}}
const g={S:{n:0,L:0,A:0,B:0,T:0},W:{n:0,L:0,A:0,B:0,T:0}};
for(let seed=0;seed<N;seed++){
  const r=rng(7000000+seed),q=init(r);let k=0;
  for(let t=0;t<PRE;t++){let b=r()<.5?1:0;k+=b;one(q,b)}
  const c=comp(q),l0=c[0]+c[1];
  const group=k===3&&l0>=.3&&l0<.35?'S':k===8&&l0>=.35&&l0<.4?'W':null;
  if(!group)continue;
  let l=l0,t=0;while(l>target&&t<MAX){both(q);let z=comp(q);l=z[0]+z[1];t++}
  let v=g[group];v.n++;v.L+=l0;v.A+=c[0];v.B+=c[1];v.T+=t;
}
for(let [name,v] of Object.entries(g))
  console.log(name,{n:v.n,initial:v.L/v.n,A:v.A/v.n,B:v.B/v.n,steps:v.T/v.n});
```

预期：S `n=9210, steps≈58.253`；W `n=2686, steps≈46.773`。

## 流式主动学习：同池 6,000 对复算

每一对 S/W 都面对同一批新候选样本、同一随机探索数及同一随机查询索引；样本输入带连续扰动，查询后不重复使用。主动选样和随机对照均按固定五点/类评价集的总体交叉熵首次达到 0.5 计时。

```js
const N=150000,PAIRS=6000,H=4,PRE=10,MAX=5000,pB=.2,eta=.12,target=.5,explore=.4,M=10;
function rng(v){let x=v>>>0;return()=>{x+=0x6D2B79F5;let t=x;t=Math.imul(t^(t>>>15),t|1);t^=t+Math.imul(t^(t>>>7),t|61);return((t^(t>>>14))>>>0)/4294967296}}
function gauss(r){return Math.sqrt(-2*Math.log(Math.max(r(),1e-12)))*Math.cos(2*Math.PI*r())}
function sig(z){return z>0?1/(1+Math.exp(-z)):Math.exp(z)/(1+Math.exp(z))}
function sp(z){return Math.max(0,z)+Math.log1p(Math.exp(-Math.abs(z)))}
function init(r){return{h:Array.from({length:H},()=>[.3*gauss(r),.3*gauss(r),.3*gauss(r)]),c:.3*gauss(r)}}
function cp(q){return{h:q.h.map(x=>[...x]),c:q.c}}
function f(q,x){let z=q.c,t=[];for(let h=0;h<H;h++){let v=Math.tanh(q.h[h][0]*x+q.h[h][1]);t.push(v);z+=q.h[h][2]*v/2}return{z,t}}
function risk(q){let a=0,b=0;for(let i=0;i<5;i++){a+=sp(f(q,.92+.04*i).z);b+=sp(-f(q,2.92+.04*i).z)}return .8*a/5+.2*b/5}
function step(q,x,y){let o=f(q,x),g=eta*(sig(o.z)-y);for(let h=0;h<H;h++){let v=q.h[h][2],t=o.t[h];q.h[h][0]-=g*v*(1-t*t)*x/2;q.h[h][1]-=g*v*(1-t*t)/2;q.h[h][2]-=g*t/2}q.c-=g}
const pools={S:[],W:[]},initial={S:0,W:0};
for(let seed=0;seed<N;seed++){
  let r=rng(41000000+seed),q=init(r),k=0;
  for(let t=0;t<PRE;t++){let y=r()<pB?1:0,x=(y?3:1)+.2*(r()-.5);k+=y;step(q,x,y)}
  let l=risk(q),group=k===0&&l>=.58&&l<.6?'S':k===4&&l>=.62&&l<.64?'W':null;
  if(group){pools[group].push(q);initial[group]+=l}
}
console.log('pools',{S:pools.S.length,W:pools.W.length,initialS:initial.S/pools.S.length,initialW:initial.W/pools.W.length});
let rr=rng(54000000),out={activeS:0,activeW:0,randomS:0,randomW:0,weakFirstActive:0,weakFirstRandom:0,fail:0};
for(let z=0;z<PAIRS;z++){
  let as=cp(pools.S[Math.floor(rr()*pools.S.length)]),aw=cp(pools.W[Math.floor(rr()*pools.W.length)]);
  let ps=cp(as),pw=cp(aw),hs=-1,hw=-1,rs=-1,rw=-1,r=rng(60000000+z);
  for(let t=1;t<=MAX&&(hs<0||hw<0||rs<0||rw<0);t++){
    let xs=[],ys=[];for(let m=0;m<M;m++){let y=r()<pB?1:0;ys.push(y);xs.push((y?3:1)+.2*(r()-.5))}
    let u=r(),v=r(),jrand=Math.floor(r()*M);
    for(let [q,name] of [[as,'hs'],[aw,'hw']])if((name==='hs'?hs:hw)<0){
      let j;if(u<explore)j=Math.floor(v*M);
      else{j=0;let best=Infinity;for(let m=0;m<M;m++){let a=Math.abs(f(q,xs[m]).z);if(a<best){best=a;j=m}}}
      step(q,xs[j],ys[j]);if(risk(q)<=target){if(name==='hs')hs=t;else hw=t}
    }
    for(let [q,name] of [[ps,'rs'],[pw,'rw']])if((name==='rs'?rs:rw)<0){
      step(q,xs[jrand],ys[jrand]);if(risk(q)<=target){if(name==='rs')rs=t;else rw=t}
    }
  }
  if(hs<0||hw<0||rs<0||rw<0)out.fail++;
  hs=hs<0?MAX:hs;hw=hw<0?MAX:hw;rs=rs<0?MAX:rs;rw=rw<0?MAX:rw;
  out.activeS+=hs;out.activeW+=hw;out.randomS+=rs;out.randomW+=rw;
  out.weakFirstActive+=hw<hs;out.weakFirstRandom+=rw<rs;
}
for(let k of Object.keys(out))if(k!=='fail')out[k]/=PAIRS;
console.log('paired',out);
```

预期：池大小 S `2349`、W `1652`；主动平均查询 S `≈158.336`、W `≈97.889`，弱者先达标 `≈0.683`；随机查询 S `≈64.656`、W `≈44.761`。

## 三层 tanh 网络：220,000 种子独立验证

梯度以中心有限差分核对，最大绝对误差低于 `6×10^-11`。下面脚本复算独立样本的组均值。

```js
const N=220000,H1=3,H2=3,PRE=10,MAX=350,s=.1,eta=.5,target=.02,r1=Math.sqrt(H1),r2=Math.sqrt(H2);
function rng(v){let x=v>>>0;return()=>{x+=0x6D2B79F5;let t=x;t=Math.imul(t^(t>>>15),t|1);t^=t+Math.imul(t^(t>>>7),t|61);return((t^(t>>>14))>>>0)/4294967296}}
function gauss(r){return Math.sqrt(-2*Math.log(Math.max(r(),1e-12)))*Math.cos(2*Math.PI*r())}
function init(r){return{w:Array.from({length:H1},()=>[.6*gauss(r),.6*gauss(r)]),U:Array.from({length:H2},()=>Array.from({length:H1},()=>.6*gauss(r))),v:Array.from({length:H2},()=>.6*gauss(r))}}
function f(q,b){let h1=q.w.map(w=>Math.tanh(b?s*w[1]:w[0])),h2=[],z=0;for(let j=0;j<H2;j++){let a=0;for(let i=0;i<H1;i++)a+=q.U[j][i]*h1[i]/r1;h2.push(Math.tanh(a));z+=q.v[j]*h2[j]/r2}return{z,h1,h2}}
function comp(q){let a=f(q,0).z-1,b=f(q,1).z-1;return [.25*a*a,.25*b*b]}
function grad(q,b){let o=f(q,b),e=o.z-1,g2=[],v=[],U=Array.from({length:H2},()=>Array(H1)),w=Array.from({length:H1},()=>[0,0]);for(let j=0;j<H2;j++){v[j]=e*o.h2[j]/r2;g2[j]=e*q.v[j]*(1-o.h2[j]**2)/r2;for(let i=0;i<H1;i++)U[j][i]=g2[j]*o.h1[i]/r1}for(let i=0;i<H1;i++){let a=0;for(let j=0;j<H2;j++)a+=g2[j]*q.U[j][i]/r1;w[i][b?1:0]=a*(1-o.h1[i]**2)*(b?s:1)}return{v,U,w}}
function update(q,a,b,scale){for(let j=0;j<H2;j++){q.v[j]-=scale*(a.v[j]+(b?.v[j]||0));for(let i=0;i<H1;i++)q.U[j][i]-=scale*(a.U[j][i]+(b?.U[j][i]||0))}for(let i=0;i<H1;i++)for(let k=0;k<2;k++)q.w[i][k]-=scale*(a.w[i][k]+(b?.w[i][k]||0))}
const g={S:{n:0,L:0,A:0,B:0,T:0},W:{n:0,L:0,A:0,B:0,T:0}};
for(let seed=0;seed<N;seed++){
  let r=rng(190000000+seed),q=init(r),k=0;
  for(let t=0;t<PRE;t++){let b=r()<.5?1:0;k+=b;update(q,grad(q,b),null,eta)}
  let c=comp(q),l0=c[0]+c[1],group=k===3&&l0>=.3&&l0<.325?'S':k===8&&l0>=.325&&l0<.35?'W':null;
  if(!group)continue;
  let l=l0,t=0;while(l>target&&t<MAX){update(q,grad(q,0),grad(q,1),eta*.5);let z=comp(q);l=z[0]+z[1];t++}
  let v=g[group];v.n++;v.L+=l0;v.A+=c[0];v.B+=c[1];v.T+=t;
}
for(let [name,v] of Object.entries(g))
  console.log(name,{n:v.n,initial:v.L/v.n,A:v.A/v.n,B:v.B/v.n,steps:v.T/v.n});
```

预期：S `n=3400, steps≈103.06`；W `n=628, steps≈85.62`。
