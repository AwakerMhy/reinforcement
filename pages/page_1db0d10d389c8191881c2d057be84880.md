# 最小复现程序

以下 JavaScript 使用 Node.js 标准库即可运行；默认仅复现严格起点排序的二维主动学习主实验。

```js
"use strict";
function rng(seed) {
  let x=seed>>>0;
  return () => {
    x=(x+0x6d2b79f5)>>>0;
    let t=Math.imul(x^(x>>>15),1|x);
    t^=t+Math.imul(t^(t>>>7),61|t);
    return ((t^(t>>>14))>>>0)/4294967296;
  };
}
function normal(u) {
  return Math.sqrt(-2*Math.log(Math.max(u(),1e-12)))*Math.cos(2*Math.PI*u());
}
function prepare(seed,pA,wB=.001) {
  const u=rng(seed),n=[0,0],sum=[0,0];
  for(let j=0;j<20;j++) {
    const k=u()<pA?0:1;
    n[k]++;sum[k]+=normal(u);
  }
  const m=sum.map((s,k)=>s/(1+n[k]));
  return {n,m,L:m[0]**2+wB*m[1]**2};
}
function future(seed) {
  const us=[rng(seed^0x12345678),rng(seed^0x91abcdef)], ys=[[],[]];
  return (k,i) => {
    while(ys[k].length<=i) ys[k].push(normal(us[k]));
    return ys[k][i];
  };
}
function run(z,y,policy,gainMode,wB=.001,goal=.01,cap=5000) {
  const m=z.m.slice(),v=z.n.map(n=>1/(1+n)),c=[0,0];
  let t=0;
  while(m[0]**2+wB*m[1]**2>goal && t<cap) {
    const k=policy==="entropy"?(v[0]>=v[1]?0:1):0;
    const g=gainMode==="bayes"?v[k]/(1+v[k]):1/(21+c[k]);
    m[k]+=g*(y(k,c[k])-m[k]);
    v[k]=1/(z.n[k]+c[k]+2);
    c[k]++;t++;
  }
  return t;
}
function report(rows,a,b,cap=5000) {
  const N=rows.length,d=rows.map(x=>x[a]-x[b]);
  const gap=d.reduce((x,y)=>x+y,0)/N;
  const se=Math.sqrt(d.reduce((x,y)=>x+(y-gap)**2,0)/(N-1)/N);
  return {
    S:rows.reduce((x,y)=>x+y[a],0)/N,
    W:rows.reduce((x,y)=>x+y[b],0)/N,
    SminusW:gap,pairedSE:se,
    Wfirst:d.filter(x=>x>0).length/N,
    censored:rows.filter(x=>x[a]===cap||x[b]===cap).length
  };
}
function expectedRisk(z,t,wB=.001) {
  const a=z.n[0],b=z.n[1],d=Math.abs(a-b);
  let ca,cb;
  if(a>=b) {
    if(t<=d) {ca=0;cb=t;}
    else {const r=t-d;ca=Math.ceil(r/2);cb=d+Math.floor(r/2);}
  } else if(t<=d) {ca=t;cb=0;}
  else {const r=t-d;ca=d+Math.ceil(r/2);cb=Math.floor(r/2);}
  const f=(k,c)=>((1+z.n[k])**2*z.m[k]**2+c)/(1+z.n[k]+c)**2;
  return f(0,ca)+wB*f(1,cb);
}
function main(nPrep=250000,nPairs=10000) {
  const S=[],W=[];
  for(let i=0;i<nPrep;i++) {
    const s=prepare((0x51100000+i*7919)>>>0,.8);
    const w=prepare((0x52200000+i*7919)>>>0,.2);
    if(s.L>=.02&&s.L<.05) S.push(s);
    if(w.L>=.08&&w.L<.15) W.push(w);
  }
  if(!S.length||!W.length) throw Error("No eligible learners");
  const select=rng(0x53300000),rows=[];
  for(let i=0;i<nPairs;i++) {
    const s=S[Math.floor(select()*S.length)];
    const w=W[Math.floor(select()*W.length)];
    const y=future((0x54400000+i*7919)>>>0);
    rows.push([
      run(s,y,"entropy","bayes"),run(w,y,"entropy","bayes"),
      run(s,y,"fixed","bayes"),run(w,y,"fixed","bayes"),
      run(s,y,"entropy","common"),run(w,y,"entropy","common"),
      run(s,y,"fixed","common"),run(w,y,"fixed","common")
    ]);
  }
  const crossing=[null,null];
  let minRiskGap=Infinity;
  for(let t=0;t<=250;t++) {
    const a=S.reduce((x,z)=>x+expectedRisk(z,t),0)/S.length;
    const b=W.reduce((x,z)=>x+expectedRisk(z,t),0)/W.length;
    if(crossing[0]===null&&a<=.01) crossing[0]=t;
    if(crossing[1]===null&&b<=.01) crossing[1]=t;
    if(t<=200) minRiskGap=Math.min(minRiskGap,b-a);
  }
  return {
    eligible:[S.length,W.length],
    initialRisk:[
      S.reduce((x,z)=>x+z.L,0)/S.length,
      W.reduce((x,z)=>x+z.L,0)/W.length
    ],
    entropyBayes:report(rows,0,1),
    fixedBayes:report(rows,2,3),
    entropyCommonGain:report(rows,4,5),
    fixedCommonGain:report(rows,6,7),
    meanRiskCrossing:crossing,
    minWminusSMeanRisk0to200:minRiskGap
  };
}
if(typeof process!=="undefined" && typeof console!=="undefined")
  console.log(JSON.stringify(main(Number(process.argv[2]||250000),
                                  Number(process.argv[3]||10000)),null,2));
```
